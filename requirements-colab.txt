# --- What DPOConfig changes out from under you ------------------------------
# Read out of trl 0.29.1's source, not recalled. DPOConfig inherits from
# TrainingArguments via trl's BaseConfig, and BOTH layers move defaults.
# Notebook 03 sets every one of these explicitly; if you adapt it, keep doing so.
#
# Changed by trl (BaseConfig):
#   gradient_checkpointing      -> True   (with PEFT this needs
#                                          enable_input_require_grads(), or the
#                                          first backward dies)
#   bf16                        -> None, resolved to `not fp16` in __post_init__
#                                          (so it switches ITSELF on, and a
#                                          Turing T4 then emulates it, slowly)
#   logging_steps               -> 10
#
# Changed by trl (DPOConfig):
#   learning_rate               -> 1e-6   (a full-fine-tune rate, far too low for
#                                          LoRA's ~0.9% of the weights)
#   loss_type                   -> ["sigmoid"], a LIST since trl 0.29. Passing
#                                          the bare string is not the same thing.
#   max_length                  -> 1024, with truncation_mode="keep_start", so
#                                          overflow eats the completion — the
#                                          only part DPO scores
#   beta                        -> 0.1
#
# Inherited unchanged from TrainingArguments, and still worth setting, because
# the defaults suit a different job:
#   per_device_train_batch_size -> 8      (one DPO example is TWO sequences, and
#                                          the reference forward doubles it again)
#   lr_scheduler_type           -> "linear"
#   save_strategy               -> "steps", every 500
#
# Note also that trl's preference collator passes `padding_side` explicitly on
# every call (left for prompts, right for completions), so it neither needs nor
# reads the padding side of the tokenizer you hand it as `processing_class`.
# Notebook 03 still left-pads, for its own batched generation.
